using BaseSpire.Slice;
using CoreTheSpire.Engine;

// Controlled room fixture: no real map generation, inventory or game RNG.
var package = new EnginePackageBuilder("example-room", "0.4.0");
var area = package.ScopeType("Area");
var runType = package.EntityType("Run");
var hp = runType.AddField("Hp", FieldTypeSchema.Int(0, 100));
var maxHp = runType.AddField("MaxHp", FieldTypeSchema.Int(1, 100));
var gold = runType.AddField("Gold", FieldTypeSchema.Int(0, 999));
var runs = package.Collection("Runs", runType);
var card = package.EntityType("Card");
var deck = package.Collection("LongTermDeck", card);
var view = package.Projection("PlayerView");
view.ParticipantRole = "player";
view.Type(runType);
var map = MapRoomWiring.Declare(package, view, runType);
var room = new RoomCompileContext(runType, runs, hp, maxHp, gold, deck, map);
ExampleRoom.Register(package, room);

var setup = package.Procedure("Setup");
var scopeSlot = setup.Local("area", FieldTypeSchema.Binding());
var runSlot = setup.Local("run", FieldTypeSchema.RefTo(runType));
var nodeSlot = setup.Local("node", FieldTypeSchema.RefTo(map.Node));
setup.Body(
    setup.CreateScope(area, scopeSlot),
    setup.CreateEntity(map.Node, setup.LocalRef(scopeSlot), nodeSlot,
        [(map.NodeType, setup.EnumConst(map.PointType, "Treasure"))], map.Nodes),
    setup.CreateEntity(runType, setup.LocalRef(scopeSlot), runSlot,
        [(hp, setup.Const(50L)), (maxHp, setup.Const(80L)), (gold, setup.Const(10L)),
         (map.RunCurrentNode, setup.LocalRef(nodeSlot)),
         (map.RunMapPhase, setup.Const((long)MapRoomWiring.TreasurePhase))], runs));
package.OnSessionStart(setup.Symbol);

var builder = new RuleSetBuilder();
builder.Add(package.Build());
var session = new Session(builder.Build(), seed: 42, participantRoles: ["player"]);
var player = session.Participant(0);
var projection = session.ProjectionsFor(player).Single();
session.Start();
Print("start");
var actions = session.LegalActions(player);
Console.WriteLine($"available_actions={actions.Count}");
if (actions.Count == 0) return;
var action = actions.Single();
session.Submit(action.Definition, player, []);
if (session.Status == SessionStatus.Faulted) throw new InvalidOperationException(session.Fault?.ToString());
Print("close");
Console.WriteLine($"remaining_actions={session.LegalActions(player).Count}");

void Print(string step)
{
    var run = session.Observe(player, projection).Instances.Single(i => i.TypeId == "example-room:Run");
    Console.WriteLine(step + ": " + string.Join(", ", run.Fields
        .Where(f => f.Name is "Gold" or "MapPhase").Select(f => $"{f.Name}={f.Value}")));
}
